```java
package com.insightflow.entity;

import jakarta.persistence.*;
import java.time.OffsetDateTime;

/**
 * 每个主题的持久化 EWMA 基线；是 Alert 计算的确定性输入。
 *
 * <p>同一 (workspace_id, issue_id) 唯一。按日增量更新：新桶到达时更新
 * EWMA + variance，记录已处理桶防幂等。基线建立期（active_buckets <
 * minHistoryDays）不触发常态告警，只标记分类。</p>
 */
@Entity
@Table(name = "issue_baseline_profile")
public class IssueBaselineProfile {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @Column(name = "workspace_id", nullable = false, updatable = false)
    private Long workspaceId;

    @Column(name = "issue_id", nullable = false, updatable = false)
    private Long issueId;

    /** 已有日桶数，满 minHistoryDays 后 status 从 baseline_building 切为 active。 */
    @Column(name = "active_buckets", nullable = false)
    private int activeBuckets;

    /** EWMA 平滑值。 */
    @Column(name = "baseline_ewma", nullable = false)
    private double baselineEwma;

    /** EWMA 方差（非标准差）。 */
    @Column(name = "baseline_variance", nullable = false)
    private double baselineVariance;

    /** 已处理的最后一个桶起点，防同日重复更新。 */
    @Column(name = "last_processed_bucket")
    private OffsetDateTime lastProcessedBucket;

    /** 最近一日的反馈数，供分类器参考。 */
    @Column(name = "last_value")
    private Integer lastValue;

    /** baseline_building / active / dormant。 */
    @Column(nullable = false, length = 30)
    private String status;

    /** surge / escalating / chronic / longtail / normal。 */
    @Column(length = 30)
    private String classification;

    @Column(name = "created_at", nullable = false, updatable = false)
    private OffsetDateTime createdAt;

    @Column(name = "updated_at", nullable = false)
    private OffsetDateTime updatedAt;

    protected IssueBaselineProfile() {}

    /** 创建全新基线；首日 EWMA = 当日值，方差 = 0，active_buckets = 1。 */
    public static IssueBaselineProfile create(Long workspaceId, Long issueId,
                                               OffsetDateTime bucketStart, int todayCount,
                                               int minHistoryDays) {
        IssueBaselineProfile p = new IssueBaselineProfile();
        OffsetDateTime now = OffsetDateTime.now();
        p.workspaceId = workspaceId;
        p.issueId = issueId;
        p.activeBuckets = 1;
        p.baselineEwma = todayCount;
        p.baselineVariance = 0;
        p.lastProcessedBucket = bucketStart;
        p.lastValue = todayCount;
        p.status = 1 < minHistoryDays ? "baseline_building" : "active";
        p.classification = "normal";
        p.createdAt = now;
        p.updatedAt = now;
        return p;
    }

    /**
     * 增量更新 EWMA + variance。
     * new_ewma = α × todayCount + (1-α) × old_ewma
     * new_variance = α × (todayCount - old_ewma)² + (1-α) × old_variance
     */
    public void updateEwma(double alpha, int todayCount, OffsetDateTime bucketStart,
                           int minHistoryDays, String newClassification) {
        double oldEwma = this.baselineEwma;
        this.baselineEwma = alpha * todayCount + (1 - alpha) * oldEwma;
        double diff = todayCount - oldEwma;
        this.baselineVariance = alpha * diff * diff + (1 - alpha) * this.baselineVariance;
        this.activeBuckets++;
        this.lastProcessedBucket = bucketStart;
        this.lastValue = todayCount;
        this.status = this.activeBuckets < minHistoryDays ? "baseline_building" : "active";
        this.classification = newClassification;
        this.updatedAt = OffsetDateTime.now();
    }

    /** 幂等守卫：同日桶已处理则跳过数值更新。 */
    public boolean isSameBucket(OffsetDateTime bucketStart) {
        return this.lastProcessedBucket != null && this.lastProcessedBucket.equals(bucketStart);
    }

    public double baselineStddev() {
        return Math.sqrt(Math.max(baselineVariance, 0));
    }

    public Long getId() { return id; }
    public Long getWorkspaceId() { return workspaceId; }
    public Long getIssueId() { return issueId; }
    public int getActiveBuckets() { return activeBuckets; }
    public double getBaselineEwma() { return baselineEwma; }
    public double getBaselineVariance() { return baselineVariance; }
    public OffsetDateTime getLastProcessedBucket() { return lastProcessedBucket; }
    public Integer getLastValue() { return lastValue; }
    public String getStatus() { return status; }
    public String getClassification() { return classification; }
    public OffsetDateTime getCreatedAt() { return createdAt; }
    public OffsetDateTime getUpdatedAt() { return updatedAt; }
}
```

**Interfaces:**
- Consumes: `IssueBaselineProfileRepository`
- Produces: `EwmaBaselineService.update(workspaceId, issueId, bucketStart, todayCount)` → `IssueBaselineProfile`（新建或更新后的 profile）

**算法（对齐 baseline.py:update_ewma + classify_issue）：**

```
update(workspaceId, issueId, bucketStart, todayCount):
  1. 查 profile
  2. 无 → create(workspaceId, issueId, bucketStart, todayCount, minHistoryDays)
  3. 有:
     a. isSameBucket → 只更新 lastValue，返回（幂等）
     b. 计算新 classification
     c. updateEwma(alpha, todayCount, bucketStart, minHistoryDays, classification)

classify(profile, todayCount):
  activeDays = profile.activeBuckets
  ewma = profile.baselineEwma
  std = max(profile.baselineStddev(), 1.0)
  z = (todayCount - ewma) / std

  if activeDays < minHistoryDays:
      if todayCount >= surgeMin: return "surge"
      if todayCount <= longtailMax: return "longtail"
      return "normal"

  if ewma >= chronicBaseline AND z < surgeZ: return "chronic"
  if z >= surgeZ AND todayCount >= surgeMin:
      return ewma >= chronicBaseline ? "escalating" : "surge"
  if todayCount <= longtailMax AND ewma < chronicBaseline: return "longtail"
  return "normal"
```

```java
package com.insightflow.service.analysis;

import com.insightflow.entity.IssueBaselineProfile;
import com.insightflow.repository.IssueBaselineProfileRepository;
import java.time.OffsetDateTime;
import org.springframework.stereotype.Component;

/**
 * 按日增量更新 EWMA 基线并分类。
 *
 * <p>算法对齐参考项目 baseline.py：update_ewma + classify_issue。
 * EWMA 平滑因子 α=0.3，基线建立期 3 天，分类为二维象限（量级 × 趋势）。</p>
 */
@Component
public class EwmaBaselineService {

    private final IssueBaselineProfileRepository profileRepository;
    private final double alpha;
    private final int minHistoryDays;
    private final double surgeZ;
    private final int surgeMin;
    private final double chronicBaseline;
    private final int longtailMax;

    public EwmaBaselineService(IssueBaselineProfileRepository profileRepository,
                                double alpha, int minHistoryDays,
                                double surgeZ, int surgeMin,
                                double chronicBaseline, int longtailMax) {
        this.profileRepository = profileRepository;
        this.alpha = alpha;
        this.minHistoryDays = minHistoryDays;
        this.surgeZ = surgeZ;
        this.surgeMin = surgeMin;
        this.chronicBaseline = chronicBaseline;
        this.longtailMax = longtailMax;
    }

    /** 按日增量更新 EWMA 基线并返回最新 profile。 */
    public IssueBaselineProfile update(Long workspaceId, Long issueId,
                                        OffsetDateTime bucketStart, int todayCount) {
        return profileRepository.findByWorkspaceIdAndIssueId(workspaceId, issueId)
                .map(existing -> {
                    if (existing.isSameBucket(bucketStart)) {
                        return existing;
                    }
                    String classification = classify(existing, todayCount);
                    existing.updateEwma(alpha, todayCount, bucketStart, minHistoryDays, classification);
                    return profileRepository.save(existing);
                })
                .orElseGet(() -> {
                    IssueBaselineProfile created = IssueBaselineProfile.create(
                            workspaceId, issueId, bucketStart, todayCount, minHistoryDays);
                    created.updateEwma(0, todayCount, bucketStart, minHistoryDays,
                            classify(created, todayCount));
                    created.getBaselineEwma(); // 触发 getter 无副作用
                    // 修正：create 已设 ewma=todayCount，不再调用 updateEwma
                    return profileRepository.save(created);
                });
    }

    /**
     * 二维象限分类（对齐 baseline.py:classify_issue）。
     *
     * @param profile 当前基线（调用前未更新）
     * @param todayCount 当日反馈数
     * @return surge / escalating / chronic / longtail / normal
     */
    public String classify(IssueBaselineProfile profile, int todayCount) {
        int activeDays = profile.getActiveBuckets();
        double ewma = profile.getBaselineEwma();
        double std = Math.max(profile.baselineStddev(), 1.0);
        double z = (todayCount - ewma) / std;

        if (activeDays < minHistoryDays) {
            if (todayCount >= surgeMin) return "surge";
            if (todayCount <= longtailMax) return "longtail";
            return "normal";
        }
        if (ewma >= chronicBaseline && z < surgeZ) return "chronic";
        if (z >= surgeZ && todayCount >= surgeMin) {
            return ewma >= chronicBaseline ? "escalating" : "surge";
        }
        if (todayCount <= longtailMax && ewma < chronicBaseline) return "longtail";
        return "normal";
    }

    public double getAlpha() { return alpha; }
    public int getMinHistoryDays() { return minHistoryDays; }
    public double getSurgeZ() { return surgeZ; }
    public int getSurgeMin() { return surgeMin; }
}
```

```java
package com.insightflow.service.analysis;

import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.insightflow.entity.Alert;
import com.insightflow.entity.IssueBaselineProfile;
import com.insightflow.repository.AlertRepository;
import java.time.Duration;
import java.time.OffsetDateTime;
import java.util.LinkedHashMap;
import java.util.Map;
import java.util.Optional;
import org.springframework.stereotype.Component;

/**
 * 基于 EWMA 基线的 z-score 突增检测；动态阈值 + 冷却期去重。
 *
 * <p>算法对齐 spike_detector.py：effectiveThreshold = max(globalThreshold, round(ewma + z × std))。
 * 冷却期由同 issue 最近一条 alert 的 created_at 判定。</p>
 */
@Component
public class AlertDetector {

    private final AlertRepository alertRepository;
    private final EwmaBaselineService ewmaBaselineService;
    private final int cooldownHours;
    private final int globalAlertThreshold;
    private final ObjectMapper objectMapper;

    public AlertDetector(AlertRepository alertRepository, EwmaBaselineService ewmaBaselineService,
                          int cooldownHours, int globalAlertThreshold, ObjectMapper objectMapper) {
        this.alertRepository = alertRepository;
        this.ewmaBaselineService = ewmaBaselineService;
        this.cooldownHours = cooldownHours;
        this.globalAlertThreshold = globalAlertThreshold;
        this.objectMapper = objectMapper;
    }

    /**
     * 检测是否需要为新桶创建告警。
     *
     * @return 新告警（需 save 到当前事务），或 empty（阈值未达/冷却期）
     */
    public Optional<Alert> detect(Long workspaceId, Long issueId, Long projectionId,
                                   OffsetDateTime bucketStart, int todayCount,
                                   IssueBaselineProfile baseline) {
        double std = baseline.baselineStddev();
        double zScore = (todayCount - baseline.getBaselineEwma()) / Math.max(std, 1.0);
        int effectiveThreshold = Math.max(globalAlertThreshold,
                (int) Math.round(baseline.getBaselineEwma() + ewmaBaselineService.getSurgeZ() * std));

        if (todayCount < effectiveThreshold) {
            return Optional.empty();
        }

        // 冷却期检查
        Optional<Alert> lastAlert = alertRepository
                .findTopByWorkspaceIdAndIssueIdOrderByCreatedAtDesc(workspaceId, issueId);
        if (lastAlert.isPresent()) {
            long hoursSince = Duration.between(lastAlert.get().getCreatedAt(), OffsetDateTime.now()).toHours();
            if (hoursSince < cooldownHours) {
                return Optional.empty();
            }
        }

        String evidenceJson = buildEvidence(bucketStart, todayCount, baseline, zScore, effectiveThreshold);
        Alert alert = Alert.active(workspaceId, issueId, projectionId, bucketStart, todayCount,
                baseline.getBaselineEwma(), std, zScore, effectiveThreshold, evidenceJson);
        return Optional.of(alertRepository.save(alert));
    }

    private String buildEvidence(OffsetDateTime bucketStart, int todayCount,
                                  IssueBaselineProfile baseline, double zScore, int effectiveThreshold) {
        Map<String, Object> evidence = new LinkedHashMap<>();
        evidence.put("bucket_start", bucketStart.toString());
        evidence.put("current_count", todayCount);
        evidence.put("baseline_ewma", baseline.getBaselineEwma());
        evidence.put("baseline_std", baseline.baselineStddev());
        evidence.put("z_score", Math.round(zScore * 100.0) / 100.0);
        evidence.put("effective_threshold", effectiveThreshold);
        evidence.put("active_buckets", baseline.getActiveBuckets());
        evidence.put("classification", baseline.getClassification());
        try {
            return objectMapper.writeValueAsString(evidence);
        } catch (JsonProcessingException e) {
            throw new IllegalStateException("Failed to serialize alert evidence", e);
        }
    }
}
```
